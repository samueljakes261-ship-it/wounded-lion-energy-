function cratePreMatchRouting(input) {
    let qs = '';
    const numbers = input.match(/\d+/g);
    if (numbers) {
        for (var i = 0; i < numbers.length; i++) {
            if (i == 0) {
                qs += 'sport=' + numbers[0];
                continue;
            } else if (i == 1) {
                qs += '&country=' + numbers[1];
                continue;
            } else if (i == 2) {
                qs += '&champ=' + numbers[2];
                continue;
            }
            else {
                qs += ',' + numbers[i];
            }
        }
    }
    return qs;
};
